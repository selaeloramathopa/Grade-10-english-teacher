CREATE TABLE public.learners (
  id uuid NOT NULL,
  first_name text NOT NULL,
  last_name text NOT NULL,
  email text NOT NULL UNIQUE,
  approved boolean DEFAULT true,
  CONSTRAINT learners_pkey PRIMARY KEY (id),
  CONSTRAINT learners_id_fkey FOREIGN KEY (id) REFERENCES auth.users(id)
);
CREATE TABLE public.teachers (
  id uuid NOT NULL,
  first_name text NOT NULL,
  last_name text NOT NULL,
  email text NOT NULL UNIQUE,
  approved boolean DEFAULT false,
  CONSTRAINT teachers_pkey PRIMARY KEY (id),
  CONSTRAINT teachers_id_fkey FOREIGN KEY (id) REFERENCES auth.users(id)
);
CREATE TABLE public.previous_papers (
  paper_id uuid NOT NULL DEFAULT uuid_generate_v4(),
  paper_title character varying NOT NULL,
  paper_description text,
  teacher_id uuid NOT NULL,
  subject character varying,
  academic_year character varying,
  exam_type character varying,
  total_questions integer,
  total_marks integer DEFAULT 100,
  duration_minutes integer,
  questions jsonb,
  file_url text,
  is_published boolean DEFAULT true,
  created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  updated_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT previous_papers_pkey PRIMARY KEY (paper_id),
  CONSTRAINT previous_papers_teacher_id_fkey FOREIGN KEY (teacher_id) REFERENCES public.teachers(id)
);
CREATE TABLE public.paper_views (
  view_id uuid NOT NULL DEFAULT uuid_generate_v4(),
  paper_id uuid NOT NULL,
  learner_id uuid NOT NULL,
  viewed_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT paper_views_pkey PRIMARY KEY (view_id),
  CONSTRAINT paper_views_paper_id_fkey FOREIGN KEY (paper_id) REFERENCES public.previous_papers(paper_id),
  CONSTRAINT paper_views_learner_id_fkey FOREIGN KEY (learner_id) REFERENCES public.learners(id)
);
CREATE TABLE public.announcements (
  announcement_id uuid NOT NULL DEFAULT uuid_generate_v4(),
  teacher_id uuid NOT NULL,
  title character varying NOT NULL,
  content text NOT NULL,
  subject character varying,
  is_pinned boolean DEFAULT false,
  created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  updated_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT announcements_pkey PRIMARY KEY (announcement_id),
  CONSTRAINT announcements_teacher_id_fkey FOREIGN KEY (teacher_id) REFERENCES public.teachers(id)
);
CREATE TABLE public.messages (
  message_id uuid NOT NULL DEFAULT uuid_generate_v4(),
  sender_id uuid NOT NULL,
  sender_type character varying NOT NULL CHECK (sender_type::text = ANY (ARRAY['teacher'::character varying, 'learner'::character varying]::text[])),
  recipient_id uuid,
  recipient_type character varying CHECK (recipient_type::text = ANY (ARRAY['teacher'::character varying, 'learner'::character varying]::text[])),
  title character varying,
  content text NOT NULL,
  message_type character varying,
  parent_message_id uuid,
  is_resolved boolean DEFAULT false,
  created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  updated_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT messages_pkey PRIMARY KEY (message_id),
  CONSTRAINT messages_parent_message_id_fkey FOREIGN KEY (parent_message_id) REFERENCES public.messages(message_id)
);
CREATE TABLE public.questions (
  question_id uuid NOT NULL DEFAULT uuid_generate_v4(),
  learner_id uuid NOT NULL,
  teacher_id uuid,
  subject character varying,
  question_text text NOT NULL,
  answer_text text,
  status character varying NOT NULL DEFAULT 'open'::character varying CHECK (status::text = ANY (ARRAY['open'::character varying, 'answered'::character varying, 'closed'::character varying]::text[])),
  asked_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  answered_at timestamp without time zone,
  created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  updated_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT questions_pkey PRIMARY KEY (question_id),
  CONSTRAINT questions_learner_id_fkey FOREIGN KEY (learner_id) REFERENCES public.learners(id),
  CONSTRAINT questions_teacher_id_fkey FOREIGN KEY (teacher_id) REFERENCES public.teachers(id)
);
CREATE TABLE public.quizzes (
  quiz_id uuid NOT NULL DEFAULT uuid_generate_v4(),
  teacher_id uuid NOT NULL,
  title character varying NOT NULL,
  description text,
  quiz_type character varying DEFAULT 'formative'::character varying,
  total_questions integer NOT NULL,
  total_points integer DEFAULT 100,
  time_limit_minutes integer,
  pass_threshold integer DEFAULT 50,
  allow_multiple_attempts boolean DEFAULT false,
  max_attempts integer DEFAULT 1,
  is_published boolean DEFAULT true,
  created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  updated_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT quizzes_pkey PRIMARY KEY (quiz_id),
  CONSTRAINT quizzes_teacher_id_fkey FOREIGN KEY (teacher_id) REFERENCES public.teachers(id)
);
CREATE TABLE public.quiz_questions (
  question_id uuid NOT NULL DEFAULT uuid_generate_v4(),
  quiz_id uuid NOT NULL,
  question_text text NOT NULL,
  options jsonb NOT NULL,
  correct_answer integer NOT NULL,
  points integer DEFAULT 1,
  created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  updated_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT quiz_questions_pkey PRIMARY KEY (question_id),
  CONSTRAINT quiz_questions_quiz_id_fkey FOREIGN KEY (quiz_id) REFERENCES public.quizzes(quiz_id)
);
CREATE TABLE public.quiz_attempts (
  attempt_id uuid NOT NULL DEFAULT uuid_generate_v4(),
  quiz_id uuid NOT NULL,
  learner_id uuid NOT NULL,
  started_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  submitted_at timestamp without time zone,
  score integer,
  status character varying NOT NULL DEFAULT 'in_progress'::character varying CHECK (status::text = ANY (ARRAY['in_progress'::character varying, 'submitted'::character varying, 'graded'::character varying]::text[])),
  created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  updated_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  total_score integer DEFAULT 0,
  CONSTRAINT quiz_attempts_pkey PRIMARY KEY (attempt_id),
  CONSTRAINT quiz_attempts_quiz_id_fkey FOREIGN KEY (quiz_id) REFERENCES public.quizzes(quiz_id),
  CONSTRAINT quiz_attempts_learner_id_fkey FOREIGN KEY (learner_id) REFERENCES public.learners(id)
);
CREATE TABLE public.quiz_answers (
  answer_id uuid NOT NULL DEFAULT uuid_generate_v4(),
  attempt_id uuid NOT NULL,
  question_id uuid NOT NULL,
  selected_option integer,
  answer_text text,
  is_correct boolean,
  points_awarded integer DEFAULT 0,
  answered_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT quiz_answers_pkey PRIMARY KEY (answer_id),
  CONSTRAINT quiz_answers_attempt_id_fkey FOREIGN KEY (attempt_id) REFERENCES public.quiz_attempts(attempt_id),
  CONSTRAINT quiz_answers_question_id_fkey FOREIGN KEY (question_id) REFERENCES public.quiz_questions(question_id)
);
